cmake_minimum_required(VERSION 3.28)
project(obs-scene-packer VERSION 0.2.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(OBS_APP "/Applications/OBS.app" CACHE PATH "Installed OBS application")
set(OBS_SOURCE "" CACHE PATH "OBS source headers, version 31.1.1 or newer")
set(OBS_RELEASE_CANDIDATE 0)
set(OBS_BETA 0)
configure_file("${OBS_SOURCE}/libobs/obsconfig.h.in" "${CMAKE_CURRENT_BINARY_DIR}/obsconfig.h")
find_package(Qt6 REQUIRED COMPONENTS Core Widgets)
add_library(packer-core STATIC src/packer.cpp)
set_target_properties(packer-core PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_link_libraries(packer-core PUBLIC $<COMPILE_ONLY:Qt6::Core>)
target_include_directories(packer-core PUBLIC src)
add_executable(packer-cli src/cli.cpp)
target_link_libraries(packer-cli PRIVATE packer-core Qt6::Core)
add_library(obs-scene-packer MODULE src/plugin-main.cpp)
target_compile_options(obs-scene-packer PRIVATE "-F${Qt6_DIR}/../../..")
target_include_directories(obs-scene-packer PRIVATE "${CMAKE_CURRENT_BINARY_DIR}" "${OBS_SOURCE}/libobs" "${OBS_SOURCE}/frontend/api" "${OBS_SOURCE}/deps")
target_link_libraries(obs-scene-packer PRIVATE packer-core $<COMPILE_ONLY:Qt6::Widgets> "${OBS_APP}/Contents/Frameworks/obs-frontend-api.dylib")
target_link_options(obs-scene-packer PRIVATE "LINKER:${OBS_APP}/Contents/Frameworks/libobs.framework/libobs" "LINKER:${OBS_APP}/Contents/Frameworks/QtCore.framework/QtCore" "LINKER:${OBS_APP}/Contents/Frameworks/QtGui.framework/QtGui" "LINKER:${OBS_APP}/Contents/Frameworks/QtWidgets.framework/QtWidgets")
set_target_properties(obs-scene-packer PROPERTIES BUNDLE TRUE BUNDLE_EXTENSION plugin PREFIX "" MACOSX_BUNDLE_INFO_PLIST "${CMAKE_CURRENT_SOURCE_DIR}/Info.plist.in" INSTALL_RPATH "@executable_path/../Frameworks" BUILD_WITH_INSTALL_RPATH TRUE)
# OBS supplies Qt at runtime; never package a second Qt runtime into the plugin.
add_custom_command(TARGET obs-scene-packer POST_BUILD COMMAND "${CMAKE_COMMAND}" -E make_directory "$<TARGET_BUNDLE_DIR:obs-scene-packer>/Contents/Resources" COMMAND "${CMAKE_COMMAND}" -E copy_if_different "${CMAKE_CURRENT_SOURCE_DIR}/README.md" "$<TARGET_BUNDLE_DIR:obs-scene-packer>/Contents/Resources/README.md" COMMAND "${CMAKE_COMMAND}" -E copy_if_different "${CMAKE_CURRENT_SOURCE_DIR}/TEMPLATE-LICENSE" "$<TARGET_BUNDLE_DIR:obs-scene-packer>/Contents/Resources/LICENSE" COMMAND /usr/bin/codesign --force --sign - "$<TARGET_BUNDLE_DIR:obs-scene-packer>")
set_property(TARGET obs-scene-packer APPEND PROPERTY LINK_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/README.md")
install(TARGETS obs-scene-packer LIBRARY DESTINATION .)
option(PACKER_BUILD_UI_TEST "Build isolated Qt host smoke test" OFF)
if(PACKER_BUILD_UI_TEST)
  add_executable(packer-ui-smoke src/ui-smoke.cpp src/plugin-main.cpp)
  target_include_directories(packer-ui-smoke PRIVATE "${CMAKE_CURRENT_BINARY_DIR}" "${OBS_SOURCE}/libobs" "${OBS_SOURCE}/frontend/api" "${OBS_SOURCE}/deps")
  target_link_libraries(packer-ui-smoke PRIVATE packer-core $<COMPILE_ONLY:Qt6::Widgets>)
  target_link_options(packer-ui-smoke PRIVATE "LINKER:${OBS_APP}/Contents/Frameworks/libobs.framework/libobs" "LINKER:${OBS_APP}/Contents/Frameworks/QtCore.framework/QtCore" "LINKER:${OBS_APP}/Contents/Frameworks/QtGui.framework/QtGui" "LINKER:${OBS_APP}/Contents/Frameworks/QtWidgets.framework/QtWidgets")
  set_target_properties(packer-ui-smoke PROPERTIES BUILD_RPATH "${OBS_APP}/Contents/Frameworks")
endif()
